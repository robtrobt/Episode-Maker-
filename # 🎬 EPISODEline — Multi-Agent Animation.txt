# 🎬 EPISODEline — Multi-Agent Animation Pipeline

Here's the full build. A modular, extensible Python pipeline with every agent, data contracts, the consistency loop, and the parallel fork/join.

---

## Project Structure

```
episodeline/
├── models.py            # Data contracts between agents
├── agents/
│   ├── base.py          # Abstract agent
│   ├── creative.py      # Creative Director
│   ├── story.py         # Story Agent
│   ├── storyboard.py    # Storyboard Agent
│   ├── character.py     # Character Agent
│   ├── world.py         # World Agent
│   ├── visual.py        # Visual Agent
│   ├── consistency.py   # Consistency Agent (with FAIL loop)
│   ├── animation.py     # Animation Agent
│   ├── audio.py         # Audio Agent
│   └── editor.py        # Editor Agent
├── pipeline.py          # Orchestrator (fork/join/loop)
├── config.py            # LLM + tool config
└── main.py              # Entry point
```

## 1. `models.py` — Data Contracts

```python
"""Data models passed between pipeline agents."""
from __future__ import annotations
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional


# ── Enums ──────────────────────────────────────────────
class Tone(Enum):
    COMEDY = "comedy"
    DRAMA = "drama"
    HORROR = "horror"
    EDUCATIONAL = "educational"
    ACTION = "action"
    WHIMSICAL = "whimsical"

class Audience(Enum):
    KIDS = "kids"
    TEENS = "teens"
    ADULTS = "adults"
    ALL_AGES = "all_ages"

class ShotType(Enum):
    WIDE = "wide"
    MEDIUM = "medium"
    CLOSE_UP = "close_up"
    EXTREME_CLOSE = "extreme_close"
    OVER_SHOULDER = "over_shoulder"
    BIRD_EYE = "bird_eye"
    LOW_ANGLE = "low_angle"

class ConsistencyVerdict(Enum):
    PASS = "pass"
    FAIL = "fail"


# ── Stage 1: Creative Brief ────────────────────────────
@dataclass
class CreativeBrief:
    raw_idea: str
    tone: Tone
    audience: Audience
    goals: list[str]
    style_notes: str = ""
    runtime_target_sec: int = 120
    aspect_ratio: str = "16:9"


# ── Stage 2: Script ───────────────────────────────────
@dataclass
class DialogueLine:
    character: str
    line: str
    emotion: str = "neutral"

@dataclass
class Script:
    title: str
    logline: str
    synopsis: str
    scenes: list[Scene] = field(default_factory=list)

@dataclass
class Scene:
    scene_id: str
    heading: str              # "INT. COFFEE SHOP - DAY"
    description: str
    dialogue: list[DialogueLine] = field(default_factory=list)
    narration: str = ""


# ── Stage 3: Storyboard ───────────────────────────────
@dataclass
class Shot:
    shot_id: str
    scene_id: str
    shot_type: ShotType
    action: str
    camera_move: str = "static"
    duration_sec: float = 3.0
    notes: str = ""

@dataclass
class Storyboard:
    shots: list[Shot] = field(default_factory=list)
    total_duration_sec: float = 0.0


# ── Stage 4a: Characters (parallel) ──────────────────
@dataclass
class CharacterSheet:
    name: str
    description: str
    pose: str = "standing neutral"
    expression: str = "neutral"
    clothing: str = ""
    color_palette: list[str] = field(default_factory=list)
    reference_tags: list[str] = field(default_factory=list)  # "iggies" anchors

@dataclass
class CharacterPack:
    characters: list[CharacterSheet] = field(default_factory=list)


# ── Stage 4b: World (parallel) ───────────────────────
@dataclass
class EnvironmentSheet:
    location: str
    description: str
    props: list[str] = field(default_factory=list)
    lighting: str = "natural daylight"
    atmosphere: str = "warm"
    color_grade: str = ""

@dataclass
class WorldPack:
    environments: list[EnvironmentSheet] = field(default_factory=list)


# ── Stage 5: Visuals ─────────────────────────────────
@dataclass
class KeyFrame:
    shot_id: str
    image_prompt: str          # fed to image gen
    layout_description: str
    character_refs: list[str] = field(default_factory=list)
    environment_ref: str = ""
    image_path: Optional[str] = None

@dataclass
class VisualPack:
    frames: list[KeyFrame] = field(default_factory=list)


# ── Stage 6: Consistency ─────────────────────────────
@dataclass
class ConsistencyIssue:
    frame_id: str
    field: str                 # "wardrobe", "face", "props"
    description: str
    severity: str = "major"    # major | minor

@dataclass
class ConsistencyReport:
    verdict: ConsistencyVerdict
    issues: list[ConsistencyIssue] = field(default_factory=list)
    revision_notes: str = ""


# ── Stage 7: Animation ───────────────────────────────
@dataclass
class AnimatedShot:
    shot_id: str
    motion_description: str
    timing_notes: str = ""
    lip_sync_script: str = ""
    transition_in: str = "cut"
    transition_out: str = "cut"
    video_path: Optional[str] = None

@dataclass
class AnimationPack:
    shots: list[AnimatedShot] = field(default_factory=list)


# ── Stage 8: Audio ────────────────────────────────────
@dataclass
class AudioTrack:
    track_type: str            # "dialogue" | "music" | "sfx" | "ambience"
    description: str
    file_path: Optional[str] = None
    start_sec: float = 0.0
    end_sec: float = 0.0

@dataclass
class AudioPack:
    tracks: list[AudioTrack] = field(default_factory=list)


# ── Stage 9: Final Edit ──────────────────────────────
@dataclass
class Episode:
    title: str
    runtime_sec: float
    video_path: Optional[str] = None
    thumbnail_path: Optional[str] = None
    metadata: dict = field(default_factory=dict)
```

## 2. `agents/base.py` — Abstract Agent

```python
"""Base class every agent inherits."""
from __future__ import annotations
import abc
import logging
from typing import Any

logger = logging.getLogger("episodeline")


class BaseAgent(abc.ABC):
    """All pipeline agents share a name, a logger, and a run() contract."""

    def __init__(self, name: str, llm_client: Any = None):
        self.name = name
        self.llm = llm_client
        self.log = logger.getChild(name)

    @abc.abstractmethod
    async def run(self, **kwargs) -> Any:
        """Execute the agent's task and return its output model."""
        ...

    def __repr__(self) -> str:
        return f"<{self.name}>"
```

### `agents/storyboard.py`
```python
from .base import BaseAgent
from models import Script, Storyboard, Shot, ShotType

class StoryboardAgent(BaseAgent):
    """Breaks the script into shots with camera direction."""

    def __init__(self, llm_client=None):
        super().__init__("Storyboard", llm_client)

    async def run(self, script: Script, **kw) -> Storyboard:
        self.log.info("Storyboarding %d scenes", len(script.scenes))
        shots = []
        for scene in script.scenes:
            shots.append(Shot(
                shot_id=f"{scene.scene_id}_01",
                scene_id=scene.scene_id,
                shot_type=ShotType.WIDE,
                action=scene.description,
                duration_sec=4.0,
            ))
            for i, d in enumerate(scene.dialogue):
                shots.append(Shot(
                    shot_id=f"{scene.scene_id}_{i+2:02d}",
                    scene_id=scene.scene_id,
                    shot_type=ShotType.CLOSE_UP,
                    action=f"{d.character}: {d.line}",
                    duration_sec=3.0,
                ))
        return Storyboard(
            shots=shots,
            total_duration_sec=sum(s.duration_sec for s in shots),
        )
```

## 4. `pipeline.py` — The Orchestrator

This is the heart. It handles **sequential flow**, **parallel fork/join**, and the **consistency FAIL loop**.

```python
"""
EPISODEline Pipeline Orchestrator

YOU → Creative → Story → Storyboard
        → fork(Character ‖ World) → join
        → Visual → Consistency ⇄ (loop)
        → Animation → Audio → Editor → 🎬
"""
from __future__ import annotations
import asyncio
import logging
from typing import Any

from models import ConsistencyVerdict
from agents.creative import CreativeDirectorAgent
from agents.story import StoryAgent
from agents.storyboard import StoryboardAgent
from agents.character import CharacterAgent
from agents.world import WorldAgent
from agents.visual import VisualAgent
from agents.consistency import ConsistencyAgent
from agents.animation import AnimationAgent
from agents.audio import AudioAgent
from agents.editor import EditorAgent

logger = logging.getLogger("episodeline.pipeline")


class Pipeline:
    def __init__(self, llm_client: Any = None):
        # Instantiate every agent
        self.creative   = CreativeDirectorAgent(llm_client)
        self.story      = StoryAgent(llm_client)
        self.storyboard = StoryboardAgent(llm_client)
        self.character  = CharacterAgent(llm_client)
        self.world      = WorldAgent(llm_client)
        self.visual     = VisualAgent(llm_client)
        self.consistency = ConsistencyAgent(llm_client)
        self.animation  = AnimationAgent(llm_client)
        self.audio      = AudioAgent(llm_client)
        self.editor     = EditorAgent(llm_client)

    async def run(self, raw_idea: str) -> "Episode":
        from models import Episode  # local to avoid circular

        logger.info("🚀 Pipeline started")
        logger.info("💡 Idea: %s", raw_idea)

        # ── 1. CREATIVE DIRECTOR ──────────────────────
        brief = await self.creative.run(raw_idea=raw_idea)
        logger.info("✅ Brief: tone=%s audience=%s", brief.tone, brief.audience)

        # ── 2. STORY ──────────────────────────────────
        script = await self.story.run(brief=brief)
        logger.info("✅ Script: '%s' (%d scenes)", script.title, len(script.scenes))

        # ── 3. STORYBOARD ────────────────────────────
        storyboard = await self.storyboard.run(script=script)
        logger.info("✅ Storyboard: %d shots, %.1fs",
                     len(storyboard.shots), storyboard.total_duration_sec)

        # ── 4. FORK: CHARACTER ‖ WORLD ───────────────
        logger.info("🔀 Forking Character + World agents")
        char_task = self.character.run(script=script, storyboard=storyboard)
        world_task = self.world.run(script=script, storyboard=storyboard)
        characters, world = await asyncio.gather(char_task, world_task)
        logger.info("✅ Characters: %d | Environments: %d",
                     len(characters.characters), len(world.environments))

        # ── 5. VISUAL ────────────────────────────────
        visuals = await self.visual.run(
            storyboard=storyboard,
            characters=characters,
            world=world,
        )
        logger.info("✅ Visuals: %d key frames", len(visuals.frames))

        # ── 6. CONSISTENCY LOOP ──────────────────────
        revision = 0
        max_revisions = self.consistency.max_revisions
        while True:
            report = await self.consistency.run(
                visuals=visuals, characters=characters,
            )
            if report.verdict == ConsistencyVerdict.PASS:
                logger.info("✅ Consistency PASS")
                break

            revision += 1
            logger.warning(
                "⚠️  Consistency FAIL (revision %d/%d): %s",
                revision, max_revisions, report.revision_notes,
            )
            if revision >= max_revisions:
                logger.warning("Max revisions reached — proceeding anyway")
                break

            # ── REVISE: send notes back to Visual Agent ──
            visuals = await self.visual.run(
                storyboard=storyboard,
                characters=characters,
                world=world,
                revision_notes=report.revision_notes,
            )

        # ── 7. ANIMATION ─────────────────────────────
        animation = await self.animation.run(
            storyboard=storyboard, visuals=visuals,
        )
        logger.info("✅ Animation: %d shots", len(animation.shots))

        # ── 8. AUDIO ─────────────────────────────────
        audio = await self.audio.run(script=script, animation=animation)
        logger.info("✅ Audio: %d tracks", len(audio.tracks))

        # ── 9. EDITOR ────────────────────────────────
        episode = await self.editor.run(
            animation=animation, audio=audio, title=script.title,
        )
        logger.info("🎬 EPISODE READY: '%s' (%.1fs)", episode.title, episode.runtime_sec)

        return episode
```

## 5. `main.py` — Entry Point

```python
"""
🎬 EPISODEline — Run the full animation pipeline.

Usage:
    python main.py "A crew of misfit gummy bears robs a candy factory"
"""
import asyncio
import logging
import sys

from pipeline import Pipeline

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s │ %(name)-22s │ %(message)s",
    datefmt="%H:%M:%S",
)


async def main():
    idea = sys.argv[1] if len(sys.argv) > 1 else (
        "A crew of misfit gummy bears robs a candy factory at midnight"
    )

    pipeline = Pipeline(llm_client=None)  # plug in your LLM here
    episode = await pipeline.run(idea)

    print("\n" + "=" * 50)
    print(f"🎬  {episode.title}")
    print(f"⏱   {episode.runtime_sec}s")
    print(f"📁  {episode.video_path}")
    print(f"📊  {episode.metadata}")
    print("=" * 50)


if __name__ == "__main__":
    asyncio.run(main())
```

## How It All Flows

```
YOU ──► "gummy bear heist"
          │
          ▼
   ┌──────────────┐
   │   CREATIVE   │ ──► Brief(tone, audience, goals)
   └──────┬───────┘
          ▼
   ┌──────────────┐
   │    STORY     │ ──► Script(scenes, dialogue)
   └──────┬───────┘
```
